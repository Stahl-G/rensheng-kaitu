import { mkdirSync, mkdtempSync, rmSync, watch, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { dirname, join } from "node:path";
import { performance } from "node:perf_hooks";

const base = mkdtempSync(join(tmpdir(), "kaitu-phone-measure-"));
process.env.KAITU_APP_HOME = join(base, "app");
process.env.KAITU_LOCAL_STATE_DIR = join(base, "state");
process.env.XDG_STATE_HOME = join(base, "xdg");
process.env.LOCALAPPDATA = join(base, "local");
process.env.KAITU_DIR = join(base, "workspace");
const core = await import("/workspace/rensheng-kaitu/packages/core/dist/index.js");
const { ulid } = await import("/workspace/rensheng-kaitu/packages/core/dist/ulid.js");
const { holdOverview } = await import("/workspace/rensheng-kaitu/packages/core/dist/overview.js");
const { deferSnapshots } = await import("/workspace/rensheng-kaitu/packages/core/dist/history.js");
const { stringifyFrontmatter } = await import("/workspace/rensheng-kaitu/packages/core/dist/frontmatter.js");
const { applyPhoneActions } = await import("/workspace/rensheng-kaitu/packages/core/dist/phone-actions.js");
const { batchSnapshot } = await import("/workspace/rensheng-kaitu/packages/core/dist/history.js");
const now = new Date("2026-10-18T08:00:00Z");
const results = {};
try {
  const ws = await core.initWorkspace(join(base, "workspace"), { now });
  const put = (file, text) => {
    const target = join(ws.root, file);
    mkdirSync(dirname(target), { recursive: true });
    writeFileSync(target, text);
  };
  for (let i = 0; i < 300; i++)
    put(
      `projects/synthetic-${i}/STATUS.md`,
      stringifyFrontmatter(
        {
          title: `合成项目 ${i}`,
          kind: i % 2 ? "work" : "personal",
          stage: "active",
          links: [],
          updated: now.toISOString(),
          next: [{ id: `next-${i}`, title: `推进合成项目 ${i}`, status: "todo" }],
        },
        "# 合成项目",
      ),
    );
  const records = Array.from({ length: 19900 }, (_, i) => ({
    id: ulid(),
    ts: "2026-10-17T08:00:00+08:00",
    by: "user",
    exercise: "合成深蹲",
    sets: 3,
    reps: 10,
    weight_kg: 30,
    session_id: `session-${i % 7}`,
  }));
  put("health/data/workouts/2026-10.jsonl", `${records.map(JSON.stringify).join("\n")}\n`);
  const tasks = Array.from({ length: 100 }, (_, i) => ({
    id: ulid(),
    ts: now.toISOString(),
    by: "user",
    title: `合成待办 ${i}`,
    status: "open",
  }));
  put("me/data/tasks/2026-10.jsonl", `${tasks.map(JSON.stringify).join("\n")}\n`);
  for (let i = 0; i < 2000; i++)
    put(
      `inbox/notes/2026-10-17/synthetic-${i}.md`,
      stringifyFrontmatter(
        {
          kind: "capture",
          agent: "你",
          date: "2026-10-17",
        },
        "合成原文内容。".repeat(20),
      ),
    );
  const data = await core.readOverviewPage(ws, now);
  const map = core.lifeMap(ws, now);
  const exploration = core.readExploration(ws);
  let start = performance.now();
  const snapshots = {
    today: core.buildPhoneToday(data, now.toISOString(), ws),
    week: core.buildPhoneWeek(data, now.toISOString()),
    map: core.buildPhoneMap(data, map, exploration, now.toISOString()),
  };
  results.projectionMs = performance.now() - start;
  for (const [name, value] of Object.entries(snapshots)) {
    const validate = core.schemas().builtin(`phone-${name}.schema.json`);
    if (!validate(value)) throw new Error(JSON.stringify(validate.errors));
    results[`${name}Bytes`] = Buffer.byteLength(`${JSON.stringify(value, null, 2)}\n`);
  }
  const remote = { inspect: async () => ({ private: true, branch: "main" }) };
  await core.connectWorkspaceBackup(ws, { repo: "synthetic/measure", text: false, records: false }, remote);
  await core.writePhoneSnapshots(ws, now);
  results.unchangedWrites = await core.writePhoneSnapshots(ws, now);
  // 与常驻应用一致：读数刷新和版本保存放后台，不计网络或 Git 工作。
  holdOverview(ws);
  deferSnapshots(ws, () => {});
  const actions = tasks.map((task) => ({
    client_id: ulid(),
    at: now.toISOString(),
    device: "3fa9c1d2",
    op: "complete_task",
    args: { id: task.id },
  }));
  const file = `inbox/agents/ios-3fa9c1d2-${ulid()}.actions.jsonl`;
  put(file, actions.map(JSON.stringify).join("\n"));
  start = performance.now();
  await core.processInbox(ws, { stableMs: 0 });
  results.actions100Ms = performance.now() - start;
  results.appliedActions = core
    .phoneReceipts(ws)
    .filter((r) => r.kind === "action" && r.status === "applied").length;
  if (results.appliedActions !== 100) throw new Error("手机操作未全部生效");
  const after = core.readRecords(ws, "task");
  await batchSnapshot(ws, "合成重复投递", () =>
    applyPhoneActions(ws, file, actions.map(JSON.stringify).join("\n")),
  );
  results.duplicateRevisions = core
    .readRecords(ws, "task")
    .reduce((n, r, i) => n + Number(r.rev !== after[i].rev), 0);
  const stop = core.startPhoneSnapshots(ws);
  start = performance.now();
  await new Promise((resolve, reject) => {
    const observer = watch(join(ws.root, "phone"), (_event, file) => {
      if (file !== "today.json") return;
      results.debounceObservedMs = performance.now() - start;
      clearTimeout(timeout);
      observer.close();
      resolve();
    });
    const timeout = setTimeout(() => {
      observer.close();
      reject(new Error("快照刷新未完成"));
    }, 15000);
    ws.writeText(
      "me/STATUS.md",
      stringifyFrontmatter(
        {
          updated: now.toISOString(),
          next: [{ id: "phone-next", title: "防抖实测下一步", status: "todo", priority: "high" }],
        },
        "",
      ),
    );
  });
  stop();
  // 可用时才测 A 任务的导出，不复制或临时实现另一分支的函数。
  if (core.listInvestments) {
    start = performance.now();
    core.listInvestments(ws);
    results.listInvestmentsMs = performance.now() - start;
  }
  if (core.completionRate) {
    const checkins = Array.from({ length: 100 }, (_, i) => ({ done: i % 3 !== 0 }));
    start = performance.now();
    for (let i = 0; i < 1000; i++) core.completionRate(checkins);
    results.completionRateMsPerCall = (performance.now() - start) / 1000;
  }
  console.log(JSON.stringify(results, null, 2));
} finally {
  rmSync(base, { recursive: true, force: true });
}
