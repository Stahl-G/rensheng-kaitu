import assert from 'node:assert/strict';
import {execFileSync} from 'node:child_process';
import {createHash} from 'node:crypto';
import {readFileSync,writeFileSync,copyFileSync,mkdirSync,mkdtempSync} from 'node:fs';
import {tmpdir} from 'node:os';
import {basename,join,resolve} from 'node:path';
const [inputArg,timezone,outArg,provenance]=process.argv.slice(2);
assert(inputArg && timezone && outArg && provenance, 'input timezone output provenance required');
const input=resolve(inputArg), name=basename(input), out=resolve(outArg);
assert(/^ios-[0-9a-f]{8}-[0-9A-HJKMNP-TV-Z]{26}\.records\.jsonl$/.test(name),'must preserve actual Swift filename');
const bytes=readFileSync(input), sha=createHash('sha256').update(bytes).digest('hex');
const root=mkdtempSync(join(tmpdir(),'kaitu-swift-meal-'));
const env={...process.env,KAITU_LOCAL_STATE_DIR:join(root,'state')};
const code='/tmp/kaitu-nutrition-integration',cli=join(code,'packages/cli/dist/main.js'),workspace=join(root,'data');
const commands=[];
function run(...args){commands.push([process.execPath,cli,...args]);return execFileSync(process.execPath,[cli,...args],{env,encoding:'utf8'});}
mkdirSync(out,{recursive:true});
writeFileSync(join(out,'init.log'),run('init',workspace,'--timezone',timezone));
copyFileSync(input,join(workspace,'inbox/agents',name));
assert.equal(createHash('sha256').update(readFileSync(join(workspace,'inbox/agents',name))).digest('hex'),sha);
writeFileSync(join(out,'consume.log'),run('phone','run','--dir',workspace));
const records=JSON.parse(run('record','list','meal','--json-out','--dir',workspace));
const today=JSON.parse(readFileSync(join(workspace,'phone/today.json'),'utf8'));
const receipts=JSON.parse(readFileSync(join(workspace,'phone/receipts.json'),'utf8'));
const ledger=JSON.parse(readFileSync(join(workspace,'.kaitu/phone-records.json'),'utf8'));
assert(ledger[name]?.consumed,'actual original file must be consumed');
assert(today.nutrition?.accounted_refs.includes(name),'same snapshot must cover original filename');
assert(receipts.items.some(r=>r.ref===name && r.status==='applied'));
const date=(value)=>new Intl.DateTimeFormat('en-CA',{timeZone:today.timezone,year:'numeric',month:'2-digit',day:'2-digit'}).format(new Date(value));
const current=records.filter(r=>date(r.ts)===today.today);
assert.equal(today.nutrition.kcal,current.reduce((s,r)=>s+r.kcal,0));
assert.equal(today.nutrition.protein_g,current.reduce((s,r)=>s+r.protein_g,0));
copyFileSync(input,join(out,name));
for(const file of ['phone/today.json','phone/receipts.json','.kaitu/phone-records.json','.kaitu/inbox-records.json'])
 copyFileSync(join(workspace,file),join(out,file.replaceAll('/','__')+'.txt'));
writeFileSync(join(out,'canonical-meals.json.txt'),JSON.stringify(records,null,2)+'\n');
const codeHash={};
for(const file of ['inbox.ts','phone-records.ts','phone-projections.ts','phone-snapshots.ts']) codeHash[file]=createHash('sha256').update(readFileSync(join(code,'packages/core/src',file))).digest('hex');
writeFileSync(join(out,'receipt.json.txt'),JSON.stringify({provenance,inputSHA256:sha,basename:name,root,timezone,today:today.today,nutrition:today.nutrition,commands,codeHash,consumer:'b21fee8 implementation plus PR62 a092270 integration; final doc-only441d839',result:'official-consumer-passed; awaiting same Swift decode'},null,2)+'\n');
console.log(JSON.stringify({out,sha,today:today.today,nutrition:today.nutrition}));
